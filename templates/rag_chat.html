// Add this to rag_chat.html async function sendChat() { const query = chatInput.value.trim(); if (!query) return; chatInput.disabled = true; sendBtn.disabled = true; chatInput.value = ''; addMessage(query, 'user', null, new Date().toLocaleTimeString()); typingIndicator.classList.add('active'); // Show what the agent is doing const queryLower = query.toLowerCase(); let statusMessage = '🔍 Analyzing your query...'; if (queryLower.includes('zone')) { statusMessage = '📍 Investigating zone data...'; } else if (queryLower.includes('violation') || queryLower.includes('violations')) { statusMessage = '🚨 Searching for violations...'; } else if (queryLower.includes('ppe') || queryLower.includes('compliance')) { statusMessage = '📊 Analyzing PPE compliance...'; } else if (queryLower.includes('manual') || queryLower.includes('policy')) { statusMessage = '📖 Searching documents...'; } typingIndicator.innerHTML = statusMessage; try { const response = await fetch('/api/agent/query', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ query: query }) }); const data = await response.json(); typingIndicator.classList.remove('active'); if (data.response) { // Show what action was taken let sourceInfo = '🤖 VisionDesk Agent'; if (data.action) { const actionLabels = { 'zone_investigation': '📍 Zone Investigation', 'recent_violations': '🚨 Violation Search', 'ppe_compliance': '📊 PPE Analysis', 'document_retrieval': '📖 Document Search', 'general_query': '📋 Dashboard Overview' }; sourceInfo += ` • ${actionLabels[data.action] || data.action}`; } addMessage(data.response, 'agent', sourceInfo, new Date().toLocaleTimeString()); } else { addMessage('❌ No response received. Please try again.', 'agent'); } } catch (error) { typingIndicator.classList.remove('active'); addMessage('❌ Error: ' + error.message, 'agent'); } chatInput.disabled = false; sendBtn.disabled = false; chatInput.focus(); }
